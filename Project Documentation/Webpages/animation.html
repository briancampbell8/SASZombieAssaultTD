Animation System

Advanced Animation Framework

System Overview

Note: The animation system provides advanced character and object animation capabilities with blend trees, state machines, and event-driven animation.

Core Features

  • Blend Trees: Smooth animation blending between multiple states
  • State Machines: Complex animation state management
  • Animation Events: Scriptable events triggered during animation playback
  • IK System: Inverse kinematics for realistic limb positioning
  • Procedural Animation: Dynamic animation generation
  • Animation Compression: Optimized animation data storage

Blend Tree System

Blend Tree Implementation

// Blend tree node public class BlendTreeNode { public string Name { get; set; } public AnimationClip Clip { get; set; } public float Weight { get; set; } public BlendTreeNode[] Children { get; set; } public BlendParameter Parameter { get; set; } public BlendType BlendType { get; set; } } // Blend tree controller public class BlendTreeController { private BlendTreeNode _rootNode; private Dictionary<string, float> _parameters = new(); public void SetParameter(string name, float value) { _parameters[name] = value; UpdateBlendWeights(); } public AnimationSample EvaluateBlendTree() { return EvaluateNode(_rootNode); } private AnimationSample EvaluateNode(BlendTreeNode node) { if (node.Clip != null) { return node.Clip.SampleAt(_parameters["Time"]); } if (node.Children.Length > 0) { return BlendChildren(node); } return AnimationSample.Default; } private AnimationSample BlendChildren(BlendTreeNode node) { var samples = new AnimationSample[node.Children.Length]; var weights = new float[node.Children.Length]; for (int i = 0; i < node.Children.Length; i++) { samples[i] = EvaluateNode(node.Children[i]); weights[i] = CalculateWeight(node.Children[i]); } return AnimationSample.Blend(samples, weights); } } // Blend types public enum BlendType { Linear1D, Linear2D, Radial2D, Direct }

State Machine System

State Machine Implementation

// Animation state machine public class AnimationStateMachine { private Dictionary<string, AnimationState> _states = new(); private AnimationState _currentState; private AnimationState _previousState; private float _transitionTime = 0f; private bool _inTransition = false; public void AddState(AnimationState state) { _states[state.Name] = state; } public void SetState(string stateName) { if (!_states.TryGetValue(stateName, out var newState)) { return; } if (_currentState != null && _currentState.Name != stateName) { StartTransition(newState); } } private void StartTransition(AnimationState newState) { _previousState = _currentState; _currentState = newState; _transitionTime = 0f; _inTransition = true; // Trigger state enter events TriggerStateEvents(newState, StateEventType.Enter); } public void Update(float deltaTime) { if (_inTransition) { UpdateTransition(deltaTime); } // Update current state UpdateCurrentState(deltaTime); } private void UpdateTransition(float deltaTime) { _transitionTime += deltaTime; var transitionDuration = GetTransitionDuration(_previousState, _currentState); if (_transitionTime >= transitionDuration) { CompleteTransition(); } } private void CompleteTransition() { _inTransition = false; _previousState = null; // Trigger state enter events TriggerStateEvents(_currentState, StateEventType.Enter); } } // State event types public enum StateEventType { Enter, Update, Exit }

Animation Events

Event System

// Animation event public struct AnimationEvent { public string Name; public float Time; public string Parameter; public object Value; public EventType Type; } // Event types public enum EventType { Trigger, Sound, Effect, Damage, Custom } // Event manager public class AnimationEventManager { private Dictionary<string, List<AnimationEvent>> _events = new(); public void RegisterEvents(string animationName, List<AnimationEvent> events) { _events[animationName] = events; } public void ProcessEvents(string animationName, float currentTime, float previousTime) { if (!_events.TryGetValue(animationName, out var events)) { return; } foreach (var evt in events) { if (ShouldTriggerEvent(evt, currentTime, previousTime)) { TriggerEvent(evt); } } } private bool ShouldTriggerEvent(AnimationEvent evt, float currentTime, float previousTime) { // Check if event time is between previous and current time return (evt.Time > previousTime && evt.Time <= currentTime) || (previousTime > currentTime && (evt.Time > previousTime || evt.Time <= currentTime)); } private void TriggerEvent(AnimationEvent evt) { switch (evt.Type) { case EventType.Trigger: EventManager.Publish(new AnimationTriggerEvent(evt.Name, evt.Parameter, evt.Value)); break; case EventType.Sound: AudioManager.PlaySound(evt.Parameter); break; case EventType.Effect: ParticleSystem.SpawnEffect(evt.Parameter); break; case EventType.Damage: DealDamage(evt.Parameter, (int)evt.Value); break; case EventType.Custom: EventManager.Publish(new CustomAnimationEvent(evt.Name, evt.Value)); break; } } }

IK System

Inverse Kinematics

// IK solver public class IKSolver { private IKChain[] _chains; private int _maxIterations = 10; private float _threshold = 0.001f; public void Solve() { foreach (var chain in _chains) { SolveChain(chain); } } private void SolveChain(IKChain chain) { for (int iteration = 0; iteration < _maxIterations; iteration++) { // FABRIK algorithm ForwardReach(chain); BackwardReach(chain); // Check convergence if (Vector3.Distance(chain.EndEffector.Position, chain.Target) < _threshold) { break; } } } private void ForwardReach(IKChain chain) { chain.Bones[chain.Bones.Length - 1].Position = chain.Target; for (int i = chain.Bones.Length - 2; i >= 0; i--) { var direction = (chain.Bones[i].Position - chain.Bones[i + 1].Position).Normalized(); chain.Bones[i].Position = chain.Bones[i + 1].Position + direction * chain.Bones[i].Length; } } private void BackwardReach(IKChain chain) { chain.Bones[0].Position = chain.RootPosition; for (int i = 1; i < chain.Bones.Length; i++) { var direction = (chain.Bones[i].Position - chain.Bones[i - 1].Position).Normalized(); chain.Bones[i].Position = chain.Bones[i - 1].Position + direction * chain.Bones[i].Length; } } } // IK chain definition public struct IKChain { public IKBone[] Bones; public Vector3 RootPosition; public Vector3 Target; public IKBone EndEffector; public float ChainLength; } // IK bone public struct IKBone { public Vector3 Position; public float Length; public Quaternion Rotation; }

Performance Optimization

Animation Compression

// Animation compression public class AnimationCompressor { public AnimationClip Compress(AnimationClip clip, CompressionSettings settings) { // Reduce keyframes var compressedKeyframes = ReduceKeyframes(clip.Keyframes, settings.Tolerance); // Quantize data compressedKeyframes = QuantizeKeyframes(compressedKeyframes, settings.Precision); // Compress rotations to quaternions compressedKeyframes = CompressRotations(compressedKeyframes); return new AnimationClip { Name = clip.Name, Duration = clip.Duration, Keyframes = compressedKeyframes, IsCompressed = true }; } private Keyframe[] ReduceKeyframes(Keyframe[] keyframes, float tolerance) { var reduced = new List<Keyframe>(); // Always keep first and last keyframes reduced.Add(keyframes[0]); for (int i = 1; i < keyframes.Length - 1; i++) { if (IsKeyframeNecessary(keyframes[i - 1], keyframes[i], keyframes[i + 1], tolerance)) { reduced.Add(keyframes[i]); } } reduced.Add(keyframes[keyframes.Length - 1]); return reduced.ToArray(); } private bool IsKeyframeNecessary(Keyframe prev, Keyframe current, Keyframe next, float tolerance) { // Check if current keyframe can be interpolated from neighbors var interpolated = InterpolateKeyframes(prev, next, 0.5f); var distance = Vector3.Distance(current.Position, interpolated.Position); return distance > tolerance; } } // Compression settings public struct CompressionSettings { public float Tolerance; public int Precision; public bool CompressRotations; public bool CompressScales; }

Character Animation Setup

// Character animation setup public class CharacterAnimationSetup { public void SetupCharacter(Entity character) { // Add animation components var animator = EntityManager.CreateEntity(); EntityManager.AddComponent(animator, new AnimationController()); EntityManager.AddComponent(animator, new Skeleton()); // Create animation states var idleState = new AnimationState { Name = "Idle", AnimationClip = "idle.anim", Loop = true, Speed = 1.0f }; var walkState = new AnimationState { Name = "Walk", AnimationClip = "walk.anim", Loop = true, Speed = 1.0f }; var runState = new AnimationState { Name = "Run", AnimationClip = "run.anim", Loop = true, Speed = 1.5f }; // Add transitions idleState.Transitions["Walk"] = new AnimationTransition { FromState = "Idle", ToState = "Walk", Duration = 0.2f, Conditions = new[] { new Condition { Parameter = "Speed", Operator = ConditionOperator.GreaterThan, Value = 0.1f } } }; // Setup state machine var stateMachine = new AnimationStateMachine(); stateMachine.AddState(idleState); stateMachine.AddState(walkState); stateMachine.AddState(runState); // Link to character EntityManager.AddComponent(character, new AnimationComponent { CurrentAnimation = "Idle", IsPlaying = true, IsLooping = true }); } }
