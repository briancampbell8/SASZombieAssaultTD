Appendix F: Testing Framework

Comprehensive Testing Strategy & Implementation

Testing Framework Overview

Note: The testing framework provides comprehensive coverage for all game systems with automated and manual testing capabilities.

Testing Pyramid

  • Unit Tests: Individual component and system testing
  • Integration Tests: System interaction testing
  • System Tests: End-to-end game system testing
  • Performance Tests: Load and stress testing
  • Manual Tests: User experience and gameplay testing

// Base test class public abstract class GameTestBase { protected TestContext Context { get; set; } protected EntityManager EntityManager { get; set; } protected SystemManager SystemManager { get; set; } [SetUp] public virtual void Setup() { Context = new TestContext(); EntityManager = new EntityManager(); SystemManager = new SystemManager(); SetupTestEnvironment(); } [TearDown] public virtual void TearDown() { CleanupTestEnvironment(); SystemManager?.Dispose(); EntityManager?.Dispose(); Context?.Dispose(); } protected virtual void SetupTestEnvironment() { } protected virtual void CleanupTestEnvironment() { } } // Example unit test [TestFixture] public class EntityComponentTests : GameTestBase { [Test] public void CreateEntity_ShouldReturnValidEntity() { // Arrange var expectedEntityCount = EntityManager.EntityCount + 1; // Act var entity = EntityManager.CreateEntity(); // Assert Assert.IsTrue(entity.IsValid); Assert.AreEqual(expectedEntityCount, EntityManager.EntityCount); } [Test] public void AddComponent_ShouldAttachToEntity() { // Arrange var entity = EntityManager.CreateEntity(); var component = new Position { X = 10, Y = 20 }; // Act EntityManager.AddComponent(entity, component); // Assert Assert.IsTrue(EntityManager.HasComponent<Position>(entity)); var retrievedComponent = EntityManager.GetComponent<Position>(entity); Assert.AreEqual(10, retrievedComponent.X); Assert.AreEqual(20, retrievedComponent.Y); } }

Performance Testing

Performance Test Framework

public class PerformanceTest { public delegate void TestAction(); public static PerformanceResult MeasurePerformance(TestAction action, int iterations = 1000) { var stopwatch = new Stopwatch(); var memoryBefore = GC.GetTotalMemory(true); stopwatch.Start(); for (int i = 0; i < iterations; i++) { action(); } stopwatch.Stop(); var memoryAfter = GC.GetTotalMemory(true); return new PerformanceResult { TotalTime = stopwatch.Elapsed, AverageTime = stopwatch.Elapsed.TotalMilliseconds / iterations, MemoryUsed = memoryAfter - memoryBefore, Iterations = iterations }; } } // Performance tests [TestFixture] public class PerformanceTests : SystemTestBase { [Test] public void EntityCreation_ShouldBeFast() { // Act var result = PerformanceTest.MeasurePerformance(() => { var entity = EntityManager.CreateEntity(); EntityManager.DestroyEntity(entity); }, 10000); // Assert Assert.Less(result.AverageTime, 0.001); // Less than 1ms per operation Assert.Less(result.MemoryUsed, 1024 * 1024); // Less than 1MB memory } [Test] public void SystemUpdate_ShouldMeetPerformanceTargets() { // Arrange var movementSystem = CreateSystem<MovementSystem>(); // Create many entities for (int i = 0; i < 1000; i++) { var entity = CreateTestEntity(new Position { X = i, Y = i }); EntityManager.AddComponent(entity, new Velocity { X = 1, Y = 1 }); } // Act var result = PerformanceTest.MeasurePerformance(() => { UpdateSystems(0.016f); }, 60); // 1 second of game time // Assert Assert.Less(result.AverageTime, 16.67); // 60 FPS target } }

Automated Testing Pipeline

Test Runner Configuration

// Test runner configuration public class TestRunnerConfig { public string TestAssemblyPath { get; set; } public string OutputDirectory { get; set; } public bool RunParallel { get; set; } = true; public int MaxParallelThreads { get; set; } = Environment.ProcessorCount; public TestFilter Filter { get; set; } public ReportingSettings Reporting { get; set; } } // Automated test runner public class AutomatedTestRunner { public TestRunResult RunTests(TestRunnerConfig config) { var results = new List<TestResult>(); var stopwatch = Stopwatch.StartNew(); // Load test assembly var assembly = Assembly.LoadFrom(config.TestAssemblyPath); // Discover tests var tests = DiscoverTests(assembly, config.Filter); // Run tests if (config.RunParallel) { results = RunTestsParallel(tests, config); } else { results = RunTestsSequential(tests, config); } stopwatch.Stop(); // Generate report var report = GenerateTestReport(results, stopwatch.Elapsed); return new TestRunResult { Results = results, TotalTime = stopwatch.Elapsed, Report = report, Success = results.All(r => r.Status == TestStatus.Passed) }; } }

Mock Framework

Mock Objects for Testing

// Mock entity manager for testing public class MockEntityManager : IEntityManager { private readonly Dictionary<Entity, Dictionary<Type, object>> _components = new(); private int _nextEntityId = 1; public Entity CreateEntity() { var entity = new Entity(_nextEntityId++); _components[entity] = new Dictionary<Type, object>(); return entity; } public void AddComponent<T>(Entity entity, T component) where T : struct { _components[entity][typeof(T)] = component; } public ref T GetComponent<T>(Entity entity) where T : struct { if (_components[entity].TryGetValue(typeof(T), out var component)) { return ref (T)component; } throw new ComponentNotFoundException(entity, typeof(T)); } public bool HasComponent<T>(Entity entity) where T : struct { return _components[entity].ContainsKey(typeof(T)); } } // Mock system for testing public class MockSystem : SystemBase { public bool Updated { get; private set; } public float LastDeltaTime { get; private set; } public override void Update() { Updated = true; LastDeltaTime = Time.DeltaTime; } public void Reset() { Updated = false; LastDeltaTime = 0; } }

Test Reporting

Test Report Generation

// Test report generator public class TestReportGenerator { public string GenerateReport(TestRunResult result) { var report = new StringBuilder(); // Header report.AppendLine("# Test Execution Report"); report.AppendLine($"Date: {DateTime.Now:yyyy-MM-dd HH:mm:ss}"); report.AppendLine($"Total Tests: {result.Results.Count}"); report.AppendLine($"Passed: {result.Results.Count(r => r.Status == TestStatus.Passed)}"); report.AppendLine($"Failed: {result.Results.Count(r => r.Status == TestStatus.Failed)}"); report.AppendLine($"Skipped: {result.Results.Count(r => r.Status == TestStatus.Skipped)}"); report.AppendLine($"Total Time: {result.TotalTime.TotalSeconds:F2}s"); report.AppendLine(); // Test results report.AppendLine("## Test Results"); foreach (var testResult in result.Results) { report.AppendLine($"- {testResult.TestName}: {testResult.Status}"); if (testResult.Status == TestStatus.Failed) { report.AppendLine($" Error: {testResult.ErrorMessage}"); } report.AppendLine($" Time: {testResult.ExecutionTime.TotalMilliseconds:F2}ms"); } // Performance summary report.AppendLine(); report.AppendLine("## Performance Summary"); var performanceTests = result.Results.Where(r => r.IsPerformanceTest); foreach (var perfTest in performanceTests) { report.AppendLine($"- {perfTest.TestName}: {perfTest.AverageTime:F4}ms avg"); } return report.ToString(); } }
