Build, Package & Distribution

Deployment Overview

Note: This guide covers the complete deployment pipeline from development to distribution.

Build Process

// Build pipeline public class BuildPipeline { public BuildResult Build(BuildSettings settings) { var result = new BuildResult(); try { // Clean output directory CleanOutputDirectory(settings.OutputDirectory); // Compile source code result.Merge(CompileSource(settings)); // Process assets result.Merge(ProcessAssets(settings)); // Create asset bundles result.Merge(CreateAssetBundles(settings)); // Generate executable result.Merge(GenerateExecutable(settings)); // Create installer result.Merge(CreateInstaller(settings)); // Validate build result.Merge(ValidateBuild(settings)); result.Success = true; } catch (Exception ex) { result.Success = false; result.Errors.Add($"Build failed: {ex.Message}"); } return result; } private BuildResult CompileSource(BuildSettings settings) { var result = new BuildResult(); // Configure compiler var compiler = new CSharpCompiler { OptimizationLevel = settings.Optimization, EnableDebugging = settings.EnableDebugging, OutputPath = Path.Combine(settings.OutputDirectory, "bin") }; // Compile assemblies var compileResult = compiler.CompileProject("SASZombieAssaultTD.csproj"); result.Merge(compileResult); return result; } }

Asset Packaging

Asset Bundle Creation

// Asset bundling system public class AssetBundler { public BundleResult CreateBundles(BuildSettings settings) { var result = new BundleResult(); // Create core bundle var coreBundle = CreateBundle("core", new[] { "Assets/Models/Player/**/*", "Assets/Textures/UI/**/*", "Assets/Audio/UI/**/*" }, settings); result.Bundles.Add(coreBundle); // Create level bundles var levelBundles = CreateLevelBundles(settings); result.Bundles.AddRange(levelBundles); // Create asset bundles var assetBundles = CreateAssetBundles(settings); result.Bundles.AddRange(assetBundles); return result; } private AssetBundle CreateBundle(string name, string[] assetPaths, BuildSettings settings) { var bundle = new AssetBundle { Name = name }; foreach (var assetPath in assetPaths) { var asset = ProcessAsset(assetPath, settings); bundle.Assets.Add(asset); } // Compress bundle bundle = CompressBundle(bundle, settings); // Save bundle var outputPath = Path.Combine(settings.OutputDirectory, "Bundles", $"{name}.bundle"); SaveBundle(bundle, outputPath); return bundle; } }

Bundle Optimization

// Bundle optimization strategies public class BundleOptimizer { public AssetBundle OptimizeBundle(AssetBundle bundle, BuildSettings settings) { // Remove unused assets bundle = RemoveUnusedAssets(bundle); // Compress textures bundle = CompressTextures(bundle, settings); // Optimize audio bundle = OptimizeAudio(bundle, settings); // Merge similar assets bundle = MergeSimilarAssets(bundle); // Generate bundle manifest bundle.Manifest = GenerateBundleManifest(bundle); return bundle; } private AssetBundle RemoveUnusedAssets(AssetBundle bundle) { var usedAssets = GetUsedAssets(bundle); var unusedAssets = bundle.Assets.Where(a => !usedAssets.Contains(a.Name)).ToList(); foreach (var unusedAsset in unusedAssets) { bundle.Assets.Remove(unusedAsset); } return bundle; } }

Version Management

Version Control Strategy

// Version management system public class VersionManager { public string CurrentVersion { get; private set; } public List<Version> VersionHistory { get; private set; } public VersionManager() { LoadVersionHistory(); CurrentVersion = GetCurrentVersion(); } public string CreateNewVersion(VersionType type, string description) { var newVersion = IncrementVersion(CurrentVersion, type); // Create version tag CreateVersionTag(newVersion, description); // Update version history VersionHistory.Add(new Version(newVersion, description, DateTime.Now)); // Save version history SaveVersionHistory(); CurrentVersion = newVersion; return newVersion; } private string IncrementVersion(string currentVersion, VersionType type) { var version = new Version(currentVersion); return type switch { VersionType.Major => $"{version.Major + 1}.0.0", VersionType.Minor => $"{version.Major}.{version.Minor + 1}.0", VersionType.Patch => $"{version.Major}.{version.Minor}.{version.Build + 1}", _ => currentVersion }; } } // Version types public enum VersionType { Major, // Breaking changes Minor, // New features Patch // Bug fixes }

Patch Management

// Patch creation system public class PatchManager { public Patch CreatePatch(string fromVersion, string toVersion) { var patch = new Patch { FromVersion = fromVersion, ToVersion = toVersion, CreatedAt = DateTime.Now }; // Compare versions var differences = CompareVersions(fromVersion, toVersion); // Create patch files patch.Files = CreatePatchFiles(differences); // Generate patch manifest patch.Manifest = GeneratePatchManifest(patch); // Compress patch patch = CompressPatch(patch); return patch; } private List<PatchFile> CreatePatchFiles(List<VersionDifference> differences) { var patchFiles = new List<PatchFile>(); foreach (var difference in differences) { var patchFile = new PatchFile { Path = difference.FilePath, Operation = difference.Type, Checksum = CalculateChecksum(difference.NewFilePath), Size = new FileInfo(difference.NewFilePath).Length }; if (difference.Type == FileOperation.Modified) { patchFile.Diff = CreateFileDiff(difference.OldFilePath, difference.NewFilePath); } patchFiles.Add(patchFile); } return patchFiles; } }
